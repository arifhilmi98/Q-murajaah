Sambungan Supabase (Pangkalan Data Awan)

Local Only

Masukkan Project URL dan Anon Public Key dari Dashboard Supabase anda untuk menyimpan data tasmik secara dalam talian (*cloud storage*).

// Salin & jalankan skrip ini dalam SQL Editor Supabase anda:

CREATE TABLE IF NOT EXISTS public.murid (
    id TEXT PRIMARY KEY,
    nama TEXT NOT NULL,
    kelas TEXT NOT NULL,
    jantina TEXT NOT NULL,
    "tahapAwal" TEXT,
    "detailAwal" TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.tasmik (
    id TEXT PRIMARY KEY,
    "muridId" TEXT REFERENCES public.murid(id) ON DELETE CASCADE,
    tarikh DATE NOT NULL,
    jenis TEXT NOT NULL,
    "bukuIqra" TEXT,
    "msIqra" INT,
    surah TEXT,
    ayat TEXT,
    juzuk INT,
    tahap TEXT NOT NULL,
    kesalahan INT DEFAULT 0,
    sasaran TEXT,
    catatan TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

ALTER TABLE public.murid ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tasmik ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public Access Murid" ON public.murid FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access Tasmik" ON public.tasmik FOR ALL USING (true) WITH CHECK (true);
